local function ObtenerTemaEspecial()
	local Fecha = os.date("*t")
	local Dia = Fecha.day
	local Mes = Fecha.month
	if Mes == 10 and Dia >= 20 then
    return Temas.Halloween
end
	if (Mes == 11 and Dia >= 25) or Mes == 12 then
    return Temas.Christmas
end
	if Mes == 1 and Dia <= 7 then
    return Temas.NewYear
end
	if Mes == 2 and Dia >= 13 and Dia <= 15 then
return Temas.Valentines
end
	if Mes == 7 and Dia == 22 then
		return Temas.Mangotuff
	end
	if Mes == 4 and Dia <= 7 then
		return Temas.Bluum3nu
	end
	if Mes == 8 and Dia == 15 then
    local Years = os.date("*t").year - 2026
    local Theme = Temas.Anniversary
    Theme.Titulo = "Fake c00lgui "..Years.." year anniversary!! 🎉"
    return Theme
end
	return Temas.Default
end
