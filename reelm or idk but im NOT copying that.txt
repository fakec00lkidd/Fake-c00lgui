local remote = tool.SyncAPI.ServerEndpoint
local function _(args)
    remote:InvokeServer(unpack(args))
end
local function CreatePart(partType, cf, parent)
    return remote:InvokeServer("CreatePart", partType or "Normal", cf, parent or workspace)
end
local function Resize(part, size, cf)
    _({"SyncResize", {{["Part"] = part, ["CFrame"] = cf, ["Size"] = size}}})
end
local function Color(part, col)
    _({"SyncColor", {{["Part"] = part, ["Color"] = col, ["UnionColoring"] = false}}})
end
local function SetAnchor(part, boolean)
    _({"SyncAnchor", {{["Part"] = part, ["Anchored"] = boolean}}})
end
local function SetLocked(part, boolean)
    _({"SetLocked", {part}, boolean})
end
local function SetName(part, stringg)
    _({"SetName", {part}, stringg})
end
local function SyncMaterial(part, mat)
    _({"SyncMaterial", {{["Part"] = part, ["Material"] = mat}}})
end
local function SetCollision(part, bool)
    _({"SyncCollision", {{["Part"] = part, ["CanCollide"] = bool}}})
end
local function AddMesh(part)
    _({"CreateMeshes", {{["Part"] = part}}})
end
local function SetMesh(part, meshid)
    _({"SyncMesh", {{["Part"] = part, ["MeshId"] = "rbxassetid://" .. meshid}}})
end
local function SetTexture(part, texid)
    _({"SyncMesh", {{["Part"] = part, ["TextureId"] = "rbxassetid://" .. texid}}})
end
local function MeshResize(part, size)
    _({"SyncMesh", {{["Part"] = part, ["Scale"] = size}}})
end
local function SpawnDecal(part, face)
    _({"CreateTextures", {{["Part"] = part, ["Face"] = face, ["TextureType"] = "Decal"}}})
end
local function AddDecal(part, asset, face)
    _({"SyncTexture", {{["Part"] = part, ["Face"] = face, ["TextureType"] = "Decal", ["Texture"] = "rbxassetid://" .. asset}}})
end
local function DestroyPart(part)
    _({"Remove", {part}})
end
local function SpawnPart(partType, partName, size, targetCF, partColor)
    local part = CreatePart(partType or "Normal", targetCF, workspace)
    if part then
        SetName(part, partName)
        Resize(part, size, targetCF)
        Color(part, partColor)
        SetAnchor(part, true)
        SetLocked(part, true)
    end
    return part
end

local GuardianRunning = true
local MapParts = {}
local function SaveChildren(part)
    local children = {}
    for _, child in ipairs(part:GetChildren()) do
        if child:IsA("Decal") then
            children[#children + 1] = {
                Type = "Decal",
                Name = child.Name,
                Face = child.Face,
                Texture = child.Texture,
                Transparency = child.Transparency,
                Color3 = child.Color3
            }
        elseif child:IsA("Texture") then
            children[#children + 1] = {
                Type = "Texture",
                Name = child.Name,
                Face = child.Face,
                Texture = child.Texture,
                Transparency = child.Transparency,
                Color3 = child.Color3,
                StudsPerTileU = child.StudsPerTileU,
                StudsPerTileV = child.StudsPerTileV,
                OffsetStudsU = child.OffsetStudsU,
                OffsetStudsV = child.OffsetStudsV
            }
        elseif child:IsA("SpecialMesh") then
            children[#children + 1] = {
                Type = "SpecialMesh",
                Name = child.Name,
                MeshType = child.MeshType,
                MeshId = child.MeshId,
                TextureId = child.TextureId,
                Scale = child.Scale,
                Offset = child.Offset
            }
        elseif child:IsA("Fire") then
            children[#children + 1] = {
                Type = "Fire",
                Name = child.Name,
                Heat = child.Heat,
                Size = child.Size,
                Color = child.Color,
                SecondaryColor = child.SecondaryColor,
                Enabled = child.Enabled
            }
        end
    end
    return children
end
local function SaveMapPart(part)
    if not part:IsA("BasePart") then
        return
    end
    MapParts[part] = {
        Name = part.Name,
        CFrame = part.CFrame,
        Size = part.Size,
        Color = part.Color,
        Material = part.Material,
        Transparency = part.Transparency,
        Reflectance = part.Reflectance,
        CanCollide = part.CanCollide,
        CanTouch = part.CanTouch,
        CanQuery = part.CanQuery,
        Anchored = part.Anchored,
        Children = SaveChildren(part)
    }
end

local function RestoreChildren(part, savedChildren)
    if not savedChildren then
        return
    end
    for _, child in ipairs(part:GetChildren()) do
        if child:IsA("Decal") or child:IsA("Texture") or child:IsA("SpecialMesh") or child:IsA("Fire") then
            child:Destroy()
        end
    end
    for _, data in ipairs(savedChildren) do
        if data.Type == "Decal" then
            local decal = Instance.new("Decal")
            decal.Name = data.Name
            decal.Face = data.Face
            decal.Texture = data.Texture
            decal.Transparency = data.Transparency
            decal.Color3 = data.Color3
            decal.Parent = part
        elseif data.Type == "Texture" then
            local texture = Instance.new("Texture")
            texture.Name = data.Name
            texture.Face = data.Face
            texture.Texture = data.Texture
            texture.Transparency = data.Transparency
            texture.Color3 = data.Color3
            texture.StudsPerTileU = data.StudsPerTileU
            texture.StudsPerTileV = data.StudsPerTileV
            texture.OffsetStudsU = data.OffsetStudsU
            texture.OffsetStudsV = data.OffsetStudsV
            texture.Parent = part
        elseif data.Type == "SpecialMesh" then
            local mesh = Instance.new("SpecialMesh")
            mesh.Name = data.Name
            mesh.MeshType = data.MeshType
            mesh.MeshId = data.MeshId
            mesh.TextureId = data.TextureId
            mesh.Scale = data.Scale
            mesh.Offset = data.Offset
            mesh.Parent = part
        elseif data.Type == "Fire" then
            local fire = Instance.new("Fire")
            fire.Name = data.Name
            fire.Heat = data.Heat
            fire.Size = data.Size
            fire.Color = data.Color
            fire.SecondaryColor = data.SecondaryColor
            fire.Enabled = data.Enabled
            fire.Parent = part
        end
    end
end
local function RecreatePart(oldPart, data)
    local newPart = remote:InvokeServer("CreatePart", "Normal", data.CFrame, workspace)
    if not newPart then
        return nil
    end
    task.wait(0.05)
    SetName(newPart, data.Name)
    Resize(newPart, data.Size, data.CFrame)
    Color(newPart, data.Color)
    SetAnchor(newPart, data.Anchored)
    SetCollision(newPart, data.CanCollide)
    SyncMaterial(newPart, data.Material)
    RestoreChildren(newPart, data.Children)
    return newPart
end
local function RemoveNewPart(part)
    if not part:IsA("BasePart") then
        return
    end
    local model = part:FindFirstAncestorOfClass("Model")
    if model and Players:GetPlayerFromCharacter(model) then
        return
    end
    DestroyPart(part)
end
local function SameCFrame(a, b)
    return (a.Position - b.Position).Magnitude < 0.001 and a.LookVector:Dot(b.LookVector) > 0.9999 and a.UpVector:Dot(b.UpVector) > 0.9999
end
local function RestoreTransparency(part, value)
    _({"SyncTransparency", {{["Part"] = part, ["Transparency"] = value}}})
end
local function RestoreReflectance(part, value)
    _({"SyncReflectance", {{["Part"] = part, ["Reflectance"] = value}}})
end
local function CheckChildren(part, savedChildren)
    local current = {}
    for _, child in ipairs(part:GetChildren()) do
        if child:IsA("Decal") or child:IsA("Texture") or child:IsA("SpecialMesh") or child:IsA("Fire") then
            current[child.Name .. "|" .. child.ClassName] = child
        end
    end
    local saved = {}
    for _, data in ipairs(savedChildren) do
        saved[data.Name .. "|" .. data.Type] = data
    end
    for key, data in pairs(saved) do
        local child = current[key]
        if not child then
            RestoreChildren(part, savedChildren)
            return
        end
        if data.Type == "Decal" then
            if child.Texture ~= data.Texture or child.Face ~= data.Face or child.Transparency ~= data.Transparency or child.Color3 ~= data.Color3 then
                RestoreChildren(part, savedChildren)
                return
            end
        elseif data.Type == "Texture" then
            if child.Texture ~= data.Texture or child.Face ~= data.Face or child.Transparency ~= data.Transparency or child.Color3 ~= data.Color3 or child.StudsPerTileU ~= data.StudsPerTileU or child.StudsPerTileV ~= data.StudsPerTileV or child.OffsetStudsU ~= data.OffsetStudsU or child.OffsetStudsV ~= data.OffsetStudsV then
                RestoreChildren(part, savedChildren)
                return
            end
        elseif data.Type == "SpecialMesh" then
            if child.MeshType ~= data.MeshType or child.MeshId ~= data.MeshId or child.TextureId ~= data.TextureId or child.Scale ~= data.Scale or child.Offset ~= data.Offset then
                RestoreChildren(part, savedChildren)
                return
            end
        elseif data.Type == "Fire" then
            if child.Heat ~= data.Heat or child.Size ~= data.Size or child.Color ~= data.Color or child.SecondaryColor ~= data.SecondaryColor or child.Enabled ~= data.Enabled then
                RestoreChildren(part, savedChildren)
                return
            end
        end
    end
    for key in pairs(current) do
        if not saved[key] then
            RestoreChildren(part, savedChildren)
            return
        end
    end
end
local function CheckPart(part, data)
    if not part.Parent then
        return
    end
    if not SameCFrame(part.CFrame, data.CFrame) then
        _({"SyncResize", {{["Part"] = part, ["CFrame"] = data.CFrame, ["Size"] = part.Size}}})
    end
    if part.Size ~= data.Size then
        _({"SyncResize", {{["Part"] = part, ["CFrame"] = data.CFrame, ["Size"] = data.Size}}})
    end
    if part.Color ~= data.Color then
        Color(part, data.Color)
    end
    if part.Material ~= data.Material then
        SyncMaterial(part, data.Material)
    end
    if part.Transparency ~= data.Transparency then
        RestoreTransparency(part, data.Transparency)
    end
    if part.Reflectance ~= data.Reflectance then
        RestoreReflectance(part, data.Reflectance)
    end
    if part.CanCollide ~= data.CanCollide then
        SetCollision(part, data.CanCollide)
    end
    if part.Anchored ~= data.Anchored then
        SetAnchor(part, data.Anchored)
    end
    if part.Name ~= data.Name then
        SetName(part, data.Name)
    end
    CheckChildren(part, data.Children)
end
local function CheckMap()
    if not GuardianRunning then
        return
    end
    for originalPart, data in pairs(MapParts) do
        if not originalPart.Parent then
            local newPart = RecreatePart(originalPart, data)
            if newPart then
                MapParts[newPart] = data
                MapParts[originalPart] = nil
            end
        else
            CheckPart(originalPart, data)
        end
    end
    for _, part in ipairs(workspace:GetDescendants()) do
        if part:IsA("BasePart") and not MapParts[part] then
            local model = part:FindFirstAncestorOfClass("Model")
            if not (model and Players:GetPlayerFromCharacter(model)) then
                RemoveNewPart(part)
            end
        end
    end
end
workspace.DescendantAdded:Connect(function(part)
    if not GuardianRunning or not part:IsA("BasePart") then
        return
    end
    task.wait()
    if MapParts[part] then
        return
    end
    local model = part:FindFirstAncestorOfClass("Model")
    if model and Players:GetPlayerFromCharacter(model) then
        return
    end
    RemoveNewPart(part)
end)
local OriginalSoundId = "rbxassetid://76237600899652"
local OriginalPitch = 0.2
local SoundCommand = ";music 76237600899652 ;volume 10 ;pitch 1.2"
local function CheckSound()
    local sound = workspace:FindFirstChild("HDAdminSound")
    if not sound or not sound:IsA("Sound") then
        RequestCommand:InvokeServer(SoundCommand)
        return
    end
    if sound.SoundId ~= OriginalSoundId then
        RequestCommand:InvokeServer(SoundCommand)
        return
    end
    if math.abs(sound.PlaybackSpeed - OriginalPitch) > 0.001 then
        RequestCommand:InvokeServer(";pitch 1.2")
    end
    if sound.Volume ~= 10 then
        RequestCommand:InvokeServer(";volume inf")
    end
end
task.spawn(function()
    while GuardianRunning do
        CheckMap()
        CheckSound()
        task.wait(0.1)
    end
end)
